PB EQU 31h
CB EQU 33h

EOI EQU 20h
IMR EQU 21h
INT0 EQU 24h
N_F10 EQU 10

ORG 40
IP_F10 dw RUT_F10

ORG 1000h
ON db 0

ORG 3000h
RUT_F10: CMP ON, 1
JZ apagar
MOV AL, 0FFh
OUT PB, AL
MOV ON, 1
MOV AL, EOI
OUT EOI, AL
IRET

apagar: MOV AL, 0
OUT PB, AL
MOV ON, 0
MOV AL, EOI
OUT EOI, AL
IRET

init: PUSH AX
CLI
MOV AL, N_F10
OUT INT0, AL
MOV AL, 11111110b
OUT IMR, AL
MOV AL, 0
OUT CB, AL
POP AX
STI
RET

ORG 2000h
CALL init
loop: JMP loop
end
