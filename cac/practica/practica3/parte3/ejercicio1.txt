PA EQU 30h
PB EQU 31h
CA EQU 32h
CB EQU 33h

EOI EQU 20h
IMR EQU 21h
INT0 EQU 24h
N_F10 EQU 10

ORG 40
IP_F10 dw RUT_F10

ORG 1000h
STR db "     "
i dw 0

ORG 3000h
RUT_F10: PUSH AX
MOV AL, 1
OUT CA, AL
MOV AL, 0
OUT CB, AL
CALL poll
MOV AL, EOI
OUT EOI, AL
POP AX
IRET

imprimir: PUSH AX
PUSH BX
MOV BX, offset STR
proximo: ADD BX, i
MOV AL, [BX]
OUT PB, AL
IN AL, PA
AND AL, 11111101b
OUT PA, AL
IN AL, PA
OR AL, 00000010b
OUT PA, AL
INC i
POP BX
POP AX
JMP poll

poll: IN AL, PA
AND AL, 1
JNZ poll
cmp i, 5
JNZ imprimir
RET

init: PUSH AX
MOV AL, N_F10
OUT INT0, AL
MOV AL, 11111110b
OUT IMR, AL
POP AX
RET

ORG 2000h
MOV AH, 5
MOV BX, offset STR
leo: INT 6
INC BX
DEC AH
JNZ leo
CALL init
loop: JMP loop
end
