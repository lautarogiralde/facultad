CONT EQU 10h
COMP EQU 11h

PA EQU 30h
PB EQU 31h
CA EQU 32h
CB EQU 33h

EOI EQU 20h
IMR EQU 21h
INT1 EQU 25h
N_TIMER EQU 10

ORG 40
IP_TIMER dw RUT_TIMER

ORG 1000h
ON DB 0

ORG 3000h
RUT_TIMER: MOV AL, 0
OUT PB, AL
MOV ON, AL
MOV AL, EOI
OUT EOI, AL
IRET

LUCES12: PUSH AX
PUSH BX
PUSH DX
MOV BX, SP
ADD BX, 8
MOV DX, [BX]
IN AL, DX
AND AL, 10000000b
JNZ prender
MOV AL, 0FFh
OUT IMR, AL
POP DX
POP BX
POP AX
RET
prender: MOV AL, 0FFh
OUT PB, AL
MOV ON, AL
MOV AL, 0
OUT CONT, AL
MOV AL, 11111101b
OUT IMR, AL
POP DX
POP BX
POP AX
RET

init: PUSH AX
MOV AL, 10000000b
OUT CA, AL
MOV AL, 0
OUT CB, AL
cli
MOV AL, 12
OUT COMP, AL
MOV AL, N_TIMER
OUT INT1, AL
MOV AL, 0FFh
OUT IMR, AL
sti
POP AX
RET

ORG 2000h
CALL init
loop: MOV DX, PA
PUSH DX
CALL LUCES12
POP DX
espero: CMP ON, 0FFh
JZ espero
JMP loop
end
