PA EQU 30h
PB EQU 31h
CA EQU 32h
CB EQU 33h

EOI EQU 20h
IMR EQU 21h
INT0 EQU 24h
N_F10 EQU 10

ORG 40
IP_F10 dw RUT_F10

ORG 1000h
ENTRADA db "PRESIONE F10 PARA COMENZAR"
MSG db ?


ORG 3000h
init: PUSH AX
MOV AL, 00000001b
OUT CA, AL
MOV AL, 0
OUT CB, AL
MOV AL, N_F10
OUT INT0, AL
MOV AL, 11111110b
OUT IMR, AL
POP AX
RET

RUT_F10: PUSH BX
PUSH CX
PUSH AX
MOV BX, offset MSG
MOV CX, 10
lectura: INT 6
INC BX
DEC CX
JNZ lectura
MOV BX, offset MSG
PUSH BX
CALL impresora
POP BX
MOV AL, EOI
OUT EOI, AL
POP AX
POP CX
POP BX
IRET

pulso: IN AL, PA
AND AL, 11111101b
OUT PA, AL
IN AL, PA
OR AL, 00000010b
OUT PA, AL
RET

poll: IN AL, PA
AND AL, 1
JNZ poll
CALL pulso
RET

impresora: MOV CX, 10
MOV BX, SP
ADD BX, 2
MOV BX, [BX]
loop: MOV AL, [BX]
OUT PB, AL
CALL poll
INC BX
DEC CX
JNZ loop
RET

ORG 2000h
CALL init
MOV BX, offset ENTRADA
MOV AL, offset MSG - offset ENTRADA
INT 7
loop2: JMP loop2
end
